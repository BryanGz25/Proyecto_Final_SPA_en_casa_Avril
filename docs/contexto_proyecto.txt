/**
 * Consulta la API GET de Mapbox SearchBox:
 * GET: https://api.mapbox.com/search/searchbox/v1/forward?q={search_text}
 */
async function buscarMapboxSearchBox(texto, proximity = null) {
  if (!texto || !texto.trim()) return [];

  const queryParams = new URLSearchParams({
    q: texto.trim(),
    access_token: MAPBOX_TOKEN,
    language: "es",
    limit: "6",
  });

  if (proximity?.lat && proximity?.lng) {
    queryParams.set("proximity", `${proximity.lng},${proximity.lat}`);
  }

  const url = `https://api.mapbox.com/search/searchbox/v1/forward?${queryParams.toString()}`;

  const res = await fetch(url);
  if (!res.ok) {
    throw new Error(`Error en Mapbox SearchBox API: ${res.status}`);
  }

  const data = await res.json();
  const features = data.features || [];

  return features.map((feature) => {
    const props = feature.properties || {};
    const coords = feature.geometry?.coordinates || [0, 0];
    const lng = coords[0];
    const lat = coords[1];

    const nombre = props.name || props.full_address || "Lugar encontrado";
    const direccionCompleta = props.full_address || props.place_formatted || nombre;
    const categoria = props.feature_type || props.maki || "Lugar";

    return {
      name: nombre,
      direccion: direccionCompleta,
      lat,
      lng,
      etiqueta: categoria,
    };
  });
}

/**
 * GeocodificaciÃ³n inversa con Mapbox para obtener el nombre de la calle/lugar
 */
async function reverseMapbox(lng, lat) {
  try {
    const url = `https://api.mapbox.com/search/searchbox/v1/reverse?longitude=${lng}&latitude=${lat}&access_token=${MAPBOX_TOKEN}&language=es&limit=1`;
    const res = await fetch(url);
    if (!res.ok) throw new Error("Fallo reverse geocoding");
    const data = await res.json();
    const feature = data.features?.[0];
    if (feature?.properties) {
      return (
        feature.properties.full_address ||
        feature.properties.place_formatted ||
        feature.properties.name
      );
    }
  } catch {
    // Silencioso fallback
  }
  return `UbicaciÃ³n (${lat.toFixed(5)}, ${lng.toFixed(5)})`;
}

const redondear = (numero) => Math.round(numero * 100) / 100;

export function construirFactura(pedido) {
  const lineas = (pedido.productos || []).map((producto) => {
    const cantidad = producto.cantidad || 1;
    const monto = redondear(producto.precio * cantidad);

    return {
      codigo: String(producto.id),
      descripcion: producto.nombre,
      cantidad,
      precioUnitario: producto.precio,
      monto,
    };
  });

  const total = redondear(
    lineas.reduce((suma, linea) => suma + linea.monto, 0)
  );

  const porcentajeIva = Number(EMISOR.porcentajeIva) || 13;
  const iva = redondear(
    (total * porcentajeIva) / (100 + porcentajeIva)
  );
  const subtotal = redondear(total - iva);

  const fecha = pedido.fecha
    ? new Date(pedido.fecha)
    : new Date();

  const numero =
    pedido.numero ||
    pedido.factura?.numero ||
    `PF-${fecha.getFullYear()}-${String(pedido.id).padStart(5, "0")}`;

  return {
    numero,
    moneda: EMISOR.moneda,
    condicionVenta: EMISOR.condicionVenta,
    medioPago: EMISOR.medioPago,
    plazoCredito: EMISOR.plazoCredito,
    porcentajeIva,
    fechaEmision: pedido.fecha || new Date().toISOString(),
    emisor: EMISOR,
    receptor: {
      nombre: pedido.cliente?.nombre || "Cliente",
      identificacion:
        pedido.cliente?.identificacion || "Sin especificar",
